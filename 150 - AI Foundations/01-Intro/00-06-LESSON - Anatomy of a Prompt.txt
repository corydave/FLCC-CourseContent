<script>
/* Resize the frame to whatever the page says it needs, so there is never a
   scrollbar inside a scrollbar, and hand the page the little context this
   window knows about itself.

   Only messages from the content host are honoured. Anything on the internet
   can postMessage into a window, and a listener that resizes a frame on
   unvalidated input will resize it to whatever it is told. */
(function () {
  var f = document.getElementById("csc150frame");
  if (!f) return;
  /* Read the origin off the iframe rather than hard-coding it: event.origin is
     scheme+host+port with no path and no trailing slash, and deriving it here
     means the two can never disagree. */
  var ORIGIN = new URL(f.src, location.href).origin;

  /* Brightspace puts the course and topic ids in its own URL. That is all the
     context available from here - replace strings like {FirstName} do NOT
     resolve inside an HTML document; they bake in whoever saved the file, so
     every learner would see that person's name. Do not use them. */
  function context() {
    var m = location.pathname.match(/\/(\d{3,})\//g) || [];
    var ids = m.map(function (x) { return x.replace(/\//g, ""); });
    return {
      tag: "csc150", type: "host", host: "brightspace",
      course: ids[0] || "", topic: ids[1] || "",
      hash: location.hash || "",
      /* Dave's call: keep our h1 even though Brightspace prints its own title
         above the frame. Set this back to true to hide the duplicate. */
      chrome: false
    };
  }

  window.addEventListener("message", function (e) {
    if (e.origin !== ORIGIN) return;
    var d = e.data;
    if (!d || d.tag !== "csc150") return;
    if (d.type === "ready") {
      f.contentWindow.postMessage(context(), ORIGIN);
    } else if (d.type === "height" && typeof d.height === "number") {
      /* Clamp. A buggy or hostile child should not be able to set a page to
         four million pixels, and a page that briefly measures near zero should
         not collapse the frame to nothing. */
      f.style.height = Math.min(Math.max(d.height, 200), 20000) + "px";
    } else if (d.type === "scroll") {
      /* The page asked to scroll to something inside itself. Only this window
         can actually move, so translate the offset and do it here. */
      var top = f.getBoundingClientRect().top + window.pageYOffset + d.top;
      if (d.block === "center") top -= (window.innerHeight - d.height) / 2;
      else top -= 16;
      window.scrollTo({ top: Math.max(0, top), behavior: d.smooth ? "smooth" : "auto" });
    }
  });
})();
</script>
